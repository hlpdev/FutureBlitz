# fb-bindgen, which turns a C header into a FutureBlitz module through libclang, built only
# where cmake/libclang.cmake finds it. It goes next to fb, which runs it for `fb bindgen`, so
# fb itself never loads libclang.
if(NOT TARGET fb_libclang)
  return()
endif()

add_library(fb_bindgen STATIC src/bindgen.cpp)
target_include_directories(fb_bindgen PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(fb_bindgen PUBLIC fb_compiler PRIVATE fb_libclang)
fb_set_warnings(fb_bindgen)

# That compiler's C standard, with GNU's extensions, as it reads C by default, and GCC's version,
# which headers ask __GNUC__ for.
set(fb_c_standard "gnu17")
if(CMAKE_C_STANDARD_COMPUTED_DEFAULT)
  set(fb_c_standard "gnu${CMAKE_C_STANDARD_COMPUTED_DEFAULT}")
endif()
set(fb_gnuc_version "")
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  set(fb_gnuc_version "${CMAKE_C_COMPILER_VERSION}")
endif()

file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/fb/bindgen_config.hpp CONTENT
"#pragma once

#include <string_view>

// Generated by CMake: how the C compiler that builds the runtime reads C. The headers it reads
// C with are the toolchain's (fb/toolchain.hpp).
namespace fb::bindgen_config {

inline constexpr std::string_view c_standard = \"${fb_c_standard}\";

// GCC's version, when that compiler is GCC.
inline constexpr std::string_view gnuc_version = \"${fb_gnuc_version}\";

} // namespace fb::bindgen_config
")

# The command line is read as fb reads its own.
add_executable(fb_bindgen_cli src/main.cpp ${PROJECT_SOURCE_DIR}/cli/src/terminal.cpp)
set_target_properties(fb_bindgen_cli PROPERTIES
  OUTPUT_NAME fb-bindgen
  RUNTIME_OUTPUT_DIRECTORY "$<TARGET_FILE_DIR:fb_cli>")
target_include_directories(fb_bindgen_cli PRIVATE ${PROJECT_SOURCE_DIR}/cli/include)
target_link_libraries(fb_bindgen_cli PRIVATE fb_bindgen)
fb_set_warnings(fb_bindgen_cli)
# fb's build puts the DLLs both need next to them.
add_dependencies(fb_bindgen_cli fb_cli)
fb_copy_libclang(fb_bindgen_cli)
