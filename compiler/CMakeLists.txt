file(GLOB_RECURSE FB_COMPILER_SOURCES CONFIGURE_DEPENDS src/*.cpp)

# toml++ reads project.toml, header-only, without exceptions, so a bad manifest is a result.
FetchContent_Declare(
  tomlplusplus
  GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git
  GIT_TAG v3.4.0
  GIT_SHALLOW TRUE
  SYSTEM
)
FetchContent_MakeAvailable(tomlplusplus)

add_library(fb_compiler STATIC ${FB_COMPILER_SOURCES})
target_include_directories(fb_compiler PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(fb_compiler PUBLIC fb_llvm PRIVATE tomlplusplus::tomlplusplus)
target_compile_definitions(fb_compiler PRIVATE TOML_EXCEPTIONS=0)
fb_set_warnings(fb_compiler)

# What the compiler needs to link programs: the target, the C++ driver that built the runtime
# (it knows where the C runtime is), lld, and the runtime itself.
if(MSVC)
  message(FATAL_ERROR "Linking programs needs a GCC or Clang toolchain for now")
endif()

execute_process(COMMAND ${CMAKE_CXX_COMPILER} -dumpmachine
  OUTPUT_VARIABLE FB_TARGET_TRIPLE OUTPUT_STRIP_TRAILING_WHITESPACE)
if(MINGW)
  # MinGW's static winpthread needs UCRT, the current C runtime, not the msvcrt it defaults to.
  set(FB_LINK_OPTIONS "\"-static\", \"-mcrtdll=ucrt\"")
elseif(APPLE)
  set(FB_LINK_OPTIONS "\"-Wl,-dead_strip\"")
else()
  set(FB_LINK_OPTIONS "\"-static-libstdc++\", \"-static-libgcc\", \"-pthread\"")
endif()

file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/fb/build_config.hpp CONTENT
"#pragma once

#include <array>
#include <string_view>

// Generated by CMake: this build's version, where its tools are, for linking programs, and where
// the standard library's source files are, which every program includes.
namespace fb::build_config {

inline constexpr std::string_view version = \"${PROJECT_VERSION}\";
inline constexpr std::string_view target_triple = \"${FB_TARGET_TRIPLE}\";
inline constexpr std::string_view linker_driver = \"${CMAKE_CXX_COMPILER}\";
inline constexpr std::string_view lld_directory = \"${FB_LLD_DIRECTORY}\";
inline constexpr std::string_view runtime_library = \"$<TARGET_FILE:fb_runtime>\";
inline constexpr std::string_view std_directory = \"${PROJECT_SOURCE_DIR}/std\";
inline constexpr auto link_options = std::to_array<std::string_view>({ ${FB_LINK_OPTIONS} });

} // namespace fb::build_config
")
