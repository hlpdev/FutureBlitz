file(GLOB_RECURSE FB_COMPILER_SOURCES CONFIGURE_DEPENDS src/*.cpp)

# toml++ reads project.toml, header-only, without exceptions, so a bad manifest is a result.
FetchContent_Declare(
  tomlplusplus
  GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git
  GIT_TAG v3.4.0
  GIT_SHALLOW TRUE
  SYSTEM
)
# fast_float reads floats, as std::from_chars does where the C++ library has it for floats; Apple's
# libc++ doesn't yet.
FetchContent_Declare(
  fast_float
  GIT_REPOSITORY https://github.com/fastfloat/fast_float.git
  GIT_TAG v8.0.2
  GIT_SHALLOW TRUE
  SYSTEM
)
FetchContent_MakeAvailable(tomlplusplus fast_float)

add_library(fb_compiler STATIC ${FB_COMPILER_SOURCES})
# fb/text_format.hpp is the runtime's, which the compiler folds constants with, so both format
# alike.
target_include_directories(fb_compiler PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated
  ${PROJECT_SOURCE_DIR}/runtime/include)
target_link_libraries(fb_compiler PUBLIC fb_llvm PRIVATE tomlplusplus::tomlplusplus FastFloat::fast_float
                      pcre2-8-static blake3)
target_compile_definitions(fb_compiler PRIVATE TOML_EXCEPTIONS=0)
fb_set_warnings(fb_compiler)

# What the compiler needs to link programs: the target, the C++ driver that built the runtime
# (it knows where the C runtime is), lld, and the runtime itself.
if(MSVC)
  message(FATAL_ERROR "Linking programs needs a GCC or Clang toolchain for now")
endif()

execute_process(COMMAND ${CMAKE_CXX_COMPILER} -dumpmachine
  OUTPUT_VARIABLE FB_TARGET_TRIPLE OUTPUT_STRIP_TRAILING_WHITESPACE)
if(MINGW)
  # MinGW's static winpthread needs UCRT, the current C runtime, not the msvcrt it defaults to.
  set(FB_LINK_OPTIONS "\"-static\", \"-mcrtdll=ucrt\"")
elseif(APPLE)
  set(FB_LINK_OPTIONS "\"-Wl,-dead_strip\"")
else()
  set(FB_LINK_OPTIONS "\"-static-libstdc++\", \"-static-libgcc\", \"-pthread\"")
endif()

# The system libraries each bundled C library needs, separated by spaces. mbedTLS's random comes
# from Windows' BCryptGenRandom.
set(FB_LIBUV_SYSTEM "$<JOIN:$<TARGET_PROPERTY:uv_a,INTERFACE_LINK_LIBRARIES>, >")
set(FB_MBEDTLS_SYSTEM "")
if(WIN32)
  set(FB_MBEDTLS_SYSTEM "bcrypt")
endif()
# libcurl's sockets and random on Windows, and the OS's roots from its store (crypt32); on macOS,
# the frameworks libcurl calls, and Security for the system's anchors. Its archives are mbedTLS's
# TLS and X.509 libraries beside its own; mbedTLS's crypto comes with mbedTLS.
set(FB_CURL_SYSTEM "")
if(WIN32)
  set(FB_CURL_SYSTEM "ws2_32 iphlpapi bcrypt advapi32 crypt32")
elseif(APPLE)
  string(JOIN " " FB_CURL_SYSTEM SystemConfiguration.framework CoreFoundation.framework
         CoreServices.framework Security.framework)
endif()
set(FB_CURL_ARCHIVES "$<TARGET_FILE:libcurl_static>;$<TARGET_FILE:mbedtls>;$<TARGET_FILE:mbedx509>")
# std.net.tls's part of mbedTLS, its TLS and X.509 libraries, which read an IP address with
# Winsock's inet_pton on Windows, and the OS's roots from Windows' store or with macOS's Security
# framework.
set(FB_TLS_SYSTEM "")
if(WIN32)
  set(FB_TLS_SYSTEM "ws2_32 crypt32")
elseif(APPLE)
  string(JOIN " " FB_TLS_SYSTEM CoreFoundation.framework Security.framework)
endif()
